{{- /* Renders one CSS sprite strip animation. Pass a dict with: src, frameWidth, frameHeight, frames, fps, alt, scale (optional 2|3). */ -}} {{- $src := .src -}} {{- $fw := .frameWidth | default 32 -}} {{- $fh := .frameHeight | default 32 -}} {{- $frames := .frames | default 1 -}} {{- $fps := .fps | default 8 -}} {{- $alt := .alt | default "" -}} {{- $scale := .scale | default "" -}} {{- $steps := cond (gt (int $frames) 1) (sub (int $frames) 1) 1 -}} {{- $class := "sprite-anim" -}} {{- if le (int $frames) 1 }}{{ $class = print $class " is-static" }}{{ end -}} {{- if eq (printf "%v" $scale) "2" }}{{ $class = print $class " is-scale-2" }}{{ end -}} {{- if eq (printf "%v" $scale) "3" }}{{ $class = print $class " is-scale-3" }}{{ end -}}
