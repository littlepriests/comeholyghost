{{- /* Figure with caption under the image. {{< figcaption src="photo.png" caption="Hello" >}} {{< figcaption src="photo.png" alt="Hello" >}}Caption text{{< /figcaption >}} src may be a page resource (same folder as the post) or a path under static/. Page resources use Fit 1600x webp q75. The master is not published. */ -}} {{- $src := .Get "src" | default (.Get 0) -}} {{- if not $src -}} {{- errorf "figcaption shortcode on %q: missing src" .Page.Path -}} {{- end -}} {{- $alt := .Get "alt" | default "" -}} {{- $caption := .Get "caption" | default (strings.TrimSpace .Inner) -}} {{- $img := false -}} {{- with .Page.Resources.GetMatch $src -}} {{- $img = . -}} {{- end -}} {{- $href := $src | relURL -}} {{- $disp := false -}} {{- if $img -}} {{- if ge $img.Width 1 -}} {{- $disp = $img.Fit "1600x1600 webp q75" -}} {{- $href = $disp.RelPermalink -}} {{- else -}} {{- $href = $img.RelPermalink -}} {{- end -}} {{- end -}} {{- if and (not $alt) $caption -}} {{- $alt = $caption | plainify -}} {{- end -}}
