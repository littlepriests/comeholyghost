{{- /* Figure with caption under the image. {{< figcaption src="photo.png" caption="Hello" >}} {{< figcaption src="photo.png" alt="Hello" >}}Caption text{{< /figcaption >}} src may be a page resource, or a path under assets/ (images/nongrid/…). Those are published at 800 and 1280. The file you dropped is not. */ -}} {{- $src := .Get "src" | default (.Get 0) -}} {{- if not $src -}} {{- errorf "figcaption shortcode on %q: missing src" .Page.Path -}} {{- end -}} {{- $alt := .Get "alt" | default "" -}} {{- $caption := .Get "caption" | default (strings.TrimSpace .Inner) -}} {{- $img := partial "image-resource.html" (dict "page" .Page "src" $src) -}} {{- $set := dict -}} {{- $href := $src | relURL -}} {{- if and $img (ge $img.Width 1) -}} {{- $set = partial "image-set.html" (dict "res" $img "widths" (slice 800 1280)) -}} {{- $href = $set.src -}} {{- else if $img -}} {{- $href = $img.RelPermalink -}} {{- end -}} {{- if and (not $alt) $caption -}} {{- $alt = $caption | plainify -}} {{- end -}}
